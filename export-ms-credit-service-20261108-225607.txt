package com.banking.credit.config;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.config.annotation.web.reactive.EnableWebFluxSecurity;
import org.springframework.security.config.web.server.ServerHttpSecurity;
import org.springframework.security.web.server.SecurityWebFilterChain;

/**
 * Seguridad del microservicio.
 * Si spring.security.dev-mode=true (default en desarrollo): permitAll().
 * Si false (produccion): OAuth2 Resource Server con JWT.
 */
@Configuration
@EnableWebFluxSecurity
public class SecurityConfig {

    @Value("${spring.security.dev-mode:false}")
    private boolean devMode;

    @Bean
    public SecurityWebFilterChain securityWebFilterChain(ServerHttpSecurity http) {
        if (devMode) {
            return http
                    .csrf(ServerHttpSecurity.CsrfSpec::disable)
                    .authorizeExchange(exchanges -> exchanges.anyExchange().permitAll())
                    .build();
        }
        return http
                .csrf(ServerHttpSecurity.CsrfSpec::disable)
                .authorizeExchange(exchanges -> exchanges.anyExchange().authenticated())
                .oauth2ResourceServer(oauth2 -> oauth2.jwt(jwt -> { }))
                .build();
    }
}

package com.banking.credit.config;

import io.netty.channel.ChannelOption;
import io.netty.handler.timeout.ReadTimeoutHandler;
import java.time.Duration;
import java.util.concurrent.TimeUnit;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.http.client.reactive.ReactorClientHttpConnector;
import org.springframework.web.reactive.function.client.WebClient;
import reactor.netty.http.client.HttpClient;

/**
 * WebClient hacia ms-transaction-service con timeout de 2 segundos
 * (regla transversal de Resilience4j / timeouts inter-servicios).
 */
@Configuration
public class WebClientConfig {

    @Bean
    public WebClient transactionWebClient(
            @Value("${transaction-service.base-url:http://ms-transaction-service}") String baseUrl) {
        var httpClient = HttpClient.create()
                .option(ChannelOption.CONNECT_TIMEOUT_MILLIS, 2000)
                .responseTimeout(Duration.ofSeconds(2))
                .doOnConnected(conn -> conn.addHandlerLast(
                        new ReadTimeoutHandler(2, TimeUnit.SECONDS)));
        return WebClient.builder()
                .baseUrl(baseUrl)
                .clientConnector(new ReactorClientHttpConnector(httpClient))
                .build();
    }
}

package com.banking.credit.controller;

import com.banking.credit.api.CreditApi;
import com.banking.credit.document.CreditDocument;
import com.banking.credit.dto.CreditCreateRequest;
import com.banking.credit.dto.CreditPage;
import com.banking.credit.dto.CreditPaymentRequest;
import com.banking.credit.dto.CreditResponse;
import com.banking.credit.dto.CreditStatus;
import com.banking.credit.dto.CreditUpdateRequest;
import com.banking.credit.dto.OperationAcceptedResponse;
import com.banking.credit.dto.OverdueStatusResponse;
import com.banking.credit.mapper.CreditMapper;
import com.banking.credit.service.CreditService;
import com.banking.credit.service.PaymentService;
import java.math.BigDecimal;
import java.util.List;
import java.util.UUID;
import lombok.RequiredArgsConstructor;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.RestController;
import org.springframework.web.server.ServerWebExchange;
import reactor.core.publisher.Mono;

/**
 * Implementacion del contrato generado (com.banking.credit.api.CreditApi).
 * Reactivo: nunca block() en el camino.
 */
@RestController
@RequiredArgsConstructor
public class CreditApiImpl implements CreditApi {

    private final CreditService creditService;
    private final PaymentService paymentService;
    private final CreditMapper creditMapper;

    @Override
    public Mono<ResponseEntity<CreditPage>> findAllCredits(String xCorrelationID, String xChannelID,
                                                           Integer page, Integer size,
                                                           ServerWebExchange exchange) {
        return creditService.findAll(page, size)
                .map(creditMapper::toResponse)
                .collectList()
                .map(items -> {
                    var p = new CreditPage();
                    p.setContent(items);
                    p.setPage(page);
                    p.setSize(size);
                    p.setTotalElements(items.size());
                    return ResponseEntity.ok(p);
                });
    }

    @Override
    public Mono<ResponseEntity<CreditResponse>> createCredit(String xCorrelationID, String xChannelID,
                                                             UUID xTransactionID,
                                                             Mono<CreditCreateRequest> body,
                                                             ServerWebExchange exchange) {
        return body.flatMap(req -> creditService.create(req.getCustomerId(),
                        req.getType().getValue(), new BigDecimal(req.getAmount()),
                        req.getCurrency(), req.getTermMonths(), xCorrelationID))
                .map(resp -> ResponseEntity.status(HttpStatus.CREATED).body(resp));
    }

    @Override
    public Mono<ResponseEntity<CreditResponse>> findCreditById(UUID creditId, String xCorrelationID,
                                                               String xChannelID,
                                                               ServerWebExchange exchange) {
        return creditService.findById(creditId)
                .map(creditMapper::toResponse)
                .map(ResponseEntity::ok);
    }

    @Override
    public Mono<ResponseEntity<CreditResponse>> updateCredit(UUID creditId, String xCorrelationID,
                                                             String xChannelID, UUID xTransactionID,
                                                             Mono<CreditUpdateRequest> body,
                                                             ServerWebExchange exchange) {
        return body.flatMap(req -> creditService.update(creditId, req, xCorrelationID))
                .map(creditMapper::toResponse)
                .map(ResponseEntity::ok);
    }

    @Override
    public Mono<ResponseEntity<Void>> deactivateCredit(UUID creditId, String xCorrelationID,
                                                       String xChannelID, UUID xTransactionID,
                                                       ServerWebExchange exchange) {
        return creditService.deactivate(creditId, xCorrelationID)
                .thenReturn(ResponseEntity.noContent().<Void>build());
    }

    @Override
    public Mono<ResponseEntity<OperationAcceptedResponse>> createCreditPayment(
            UUID creditId, String xCorrelationID, String xChannelID, UUID xTransactionID,
            Mono<CreditPaymentRequest> body, ServerWebExchange exchange) {
        return body.flatMap(req -> paymentService.pay(creditId, xTransactionID,
                        new BigDecimal(req.getAmount()), req.getCurrency(),
                        req.getSourceAccountId(), req.getPayerCustomerId(), xCorrelationID))
                .map(resp -> ResponseEntity.status(HttpStatus.ACCEPTED).body(resp));
    }

    @Override
    public Mono<ResponseEntity<OverdueStatusResponse>> getOverdueStatus(
            UUID creditId, String xCorrelationID, String xChannelID,
            ServerWebExchange exchange) {
        return creditService.getOverdueInfo(creditId)
                .map(info -> {
                    var r = new OverdueStatusResponse();
                    r.setCreditId(creditId);
                    r.setHasOverdueDebt(info.hasOverdueDebt());
                    r.setOverdueAmount(info.overdueAmount().toPlainString());
                    r.setDaysOverdue((int) info.daysOverdue());
                    return ResponseEntity.ok(r);
                });
    }
}

package com.banking.credit.event;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.kafka.core.reactive.ReactiveKafkaProducerTemplate;
import org.springframework.stereotype.Component;
import reactor.core.publisher.Mono;

/**
 * Productor de eventos de dominio hacia el topico banking.credit.events.v1.
 */
@Slf4j
@Component
@RequiredArgsConstructor
public class CreditEventPublisher {

    public static final String TOPIC = "banking.credit.events.v1";

    private final ReactiveKafkaProducerTemplate<String, CreditEvent> kafkaTemplate;

    public Mono<Void> publish(CreditEvent event) {
        return kafkaTemplate.send(TOPIC, event.getCreditId().toString(), event)
                .doOnSuccess(result -> log.info(
                        "Event {} published for credit {} [correlationId={}]",
                        event.getEventType(), event.getCreditId(), event.getCorrelationId()))
                .doOnError(err -> log.error(
                        "Failed to publish event {} for credit {}: {}",
                        event.getEventType(), event.getCreditId(), err.getMessage()))
                .onErrorResume(err -> Mono.empty())
                .then();
    }
}

    }

package com.banking.credit.exception;

import com.banking.credit.dto.ErrorResponse;
import java.time.OffsetDateTime;
import lombok.extern.slf4j.Slf4j;
import org.springframework.boot.autoconfigure.web.WebProperties;
import org.springframework.boot.autoconfigure.web.reactive.error.AbstractErrorWebExceptionHandler;
import org.springframework.boot.web.error.ErrorAttributeOptions;
import org.springframework.boot.web.reactive.error.ErrorAttributes;
import org.springframework.context.ApplicationContext;
import org.springframework.http.HttpStatus;
import org.springframework.http.MediaType;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.bind.support.WebExchangeBindException;
import org.springframework.web.reactive.function.server.RequestPredicates;
import org.springframework.web.reactive.function.server.RouterFunction;
import org.springframework.web.reactive.function.server.RouterFunctions;
import org.springframework.web.reactive.function.server.ServerRequest;
import org.springframework.web.reactive.function.server.ServerResponse;
import org.springframework.web.server.ServerWebExchange;
import reactor.core.publisher.Mono;

/**
 * Manejo de errores comun con estructura ErrorResponse del contrato.
 */
@Slf4j
@RestControllerAdvice
public class GlobalExceptionHandler {

    private static ErrorResponse build(String code, String message, String correlationId,
                                       java.util.List<String> details) {
        var e = new ErrorResponse();
        e.setCode(code);
        e.setMessage(message);
        e.setTimestamp(OffsetDateTime.now());
        e.setCorrelationId(correlationId);
        e.setDetails(details);
        return e;
    }

    private static String correlation(ServerWebExchange exchange) {
        var v = exchange.getRequest().getHeaders().getFirst("X-Correlation-ID");
        return v == null ? "" : v;
    }

    @ExceptionHandler(CreditNotFoundException.class)
    public Mono<ResponseEntity<ErrorResponse>> notFound(CreditNotFoundException ex,
                                                        ServerWebExchange exchange) {
        return Mono.just(ResponseEntity.status(HttpStatus.NOT_FOUND)
                .body(build(ex.getCode(), ex.getMessage(), correlation(exchange), null)));
    }

    @ExceptionHandler(CreditConflictException.class)
    public Mono<ResponseEntity<ErrorResponse>> conflict(CreditConflictException ex,
                                                        ServerWebExchange exchange) {
        return Mono.just(ResponseEntity.status(HttpStatus.CONFLICT)
                .body(build(ex.getCode(), ex.getMessage(), correlation(exchange), null)));
    }

    @ExceptionHandler(PaymentExceedsBalanceException.class)
    public Mono<ResponseEntity<ErrorResponse>> unprocessable(PaymentExceedsBalanceException ex,
                                                             ServerWebExchange exchange) {
        return Mono.just(ResponseEntity.status(HttpStatus.UNPROCESSABLE_ENTITY)
                .body(build(ex.getCode(), ex.getMessage(), correlation(exchange), null)));
    }

    @ExceptionHandler(TransactionServiceException.class)
    public Mono<ResponseEntity<ErrorResponse>> transactionError(TransactionServiceException ex,
                                                                ServerWebExchange exchange) {
        return Mono.just(ResponseEntity.status(HttpStatus.BAD_GATEWAY)
                .body(build(ex.getCode(), ex.getMessage(), correlation(exchange), null)));
    }

    @ExceptionHandler(WebExchangeBindException.class)
    public Mono<ResponseEntity<ErrorResponse>> badRequest(WebExchangeBindException ex,
                                                          ServerWebExchange exchange) {
        var details = ex.getBindingResult().getFieldErrors().stream()
                .map(fe -> fe.getField() + ": " + fe.getDefaultMessage())
                .toList();
        return Mono.just(ResponseEntity.badRequest()
                .body(build("BAD_REQUEST", "Invalid request", correlation(exchange), details)));
    }

    @ExceptionHandler(Exception.class)
    public Mono<ResponseEntity<ErrorResponse>> generic(Exception ex, ServerWebExchange exchange) {
        log.error("Unhandled error", ex);
        return Mono.just(ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR)
                .body(build("INTERNAL_ERROR", "Internal server error", correlation(exchange), null)));
    }
}

package com.banking.credit.service.impl;

import com.banking.credit.document.CreditDocument;
import com.banking.credit.event.CreditEvent;
import com.banking.credit.dto.CreditResponse;
import com.banking.credit.event.CreditEventPublisher;
import com.banking.credit.exception.CreditConflictException;
import com.banking.credit.exception.CreditNotFoundException;
import com.banking.credit.mapper.CreditMapper;
import com.banking.credit.repository.CreditRepository;
import com.banking.credit.service.CreditService;
import java.math.BigDecimal;
import java.time.OffsetDateTime;
import java.time.ZoneOffset;
import java.time.temporal.ChronoUnit;
import java.util.UUID;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import reactor.core.publisher.Flux;
import reactor.core.publisher.Mono;

/**
 * Implementacion reactiva (Mono/Flux) de las reglas de negocio de creditos.
 */
@Slf4j
@Service
@RequiredArgsConstructor
public class CreditServiceImpl implements CreditService {

    public static final String TYPE_PERSONAL = "PERSONAL";
    public static final String TYPE_BUSINESS = "BUSINESS";

    private final CreditRepository creditRepository;
    private final CreditMapper creditMapper;
    private final CreditEventPublisher eventPublisher;

    @Override
    public Mono<CreditResponse> create(UUID customerId, String type, BigDecimal amount,
                                       String currency, Integer termMonths, String correlationId) {
        return validateType(type)
                // Consultamos de forma segura (solo un booleano) si ya existe
                .then(creditRepository.existsByCustomerIdAndType(customerId, type))
                .flatMap(exists -> {
                    // Regla de Negocio: Bloquear solo si es PERSONAL y ya tiene uno
                    if (TYPE_PERSONAL.equals(type) && Boolean.TRUE.equals(exists)) {
                        return Mono.error(new CreditConflictException(
                                "Customer already has a PERSONAL credit"));
                    }
                    // Si es BUSINESS, ignoramos si existe y permitimos crear otro
                    var doc = creditMapper.toNewDocument(customerId, type, amount, currency, termMonths);
                    return creditRepository.save(doc);
                })
                .flatMap(saved -> eventPublisher.publish(CreditEvent.builder()
                                .eventType(CreditEvent.CREATED)
                                .creditId(saved.getId())
                                .customerId(saved.getCustomerId())
                                .status(saved.getStatus())
                                .balance(saved.getBalance().toPlainString())
                                .currency(saved.getCurrency())
                                .occurredAt(OffsetDateTime.now(ZoneOffset.UTC))
                                .correlationId(correlationId)
                                .build())
                        .thenReturn(saved))
                .map(creditMapper::toResponse);
    }

    @Override
    public Flux<CreditDocument> findAll(int page, int size) {
        return creditRepository.findAll()
                .skip((long) page * size)
                .take(size);
    }

    @Override
    public Mono<CreditDocument> findById(UUID creditId) {
        return creditRepository.findById(creditId)
                .switchIfEmpty(Mono.error(new CreditNotFoundException(
                        "Credit not found: " + creditId)));
    }

    @Override
    public Mono<CreditDocument> update(UUID creditId, com.banking.credit.dto.CreditUpdateRequest request,
                                       String correlationId) {
        return findById(creditId)
                .flatMap(doc -> {
                    var newStatus = request.getStatus() == null ? null : request.getStatus().getValue();
                    if (newStatus == null) {
                        return Mono.just(doc);
                    }
                    // No se permite PAID por PATCH si balance > 0 (regla de negocio).
                    if (CreditDocument.STATUS_PAID.equals(newStatus)
                            && doc.getBalance().compareTo(BigDecimal.ZERO) > 0) {
                        return Mono.error(new CreditConflictException(
                                "Cannot set PAID while balance is greater than zero"));
                    }
                    doc.setStatus(newStatus);
                    doc.setUpdatedAt(OffsetDateTime.now(ZoneOffset.UTC).toLocalDateTime());
                    return creditRepository.save(doc)
                            .flatMap(saved -> eventPublisher.publish(CreditEvent.builder()
                                            .eventType(CreditEvent.UPDATED)
                                            .creditId(saved.getId())
                                            .customerId(saved.getCustomerId())
                                            .status(saved.getStatus())
                                            .balance(saved.getBalance().toPlainString())
                                            .currency(saved.getCurrency())
                                            .occurredAt(OffsetDateTime.now(ZoneOffset.UTC))
                                            .correlationId(correlationId)
                                            .build())
                                    .thenReturn(saved));
                });
    }

    @Override
    public Mono<Void> deactivate(UUID creditId, String correlationId) {
        return findById(creditId)
                .flatMap(doc -> {
                    doc.setStatus(CreditDocument.STATUS_CANCELLED);
                    doc.setUpdatedAt(OffsetDateTime.now(ZoneOffset.UTC).toLocalDateTime());
                    return creditRepository.save(doc)
                            .flatMap(saved -> eventPublisher.publish(CreditEvent.builder()
                                            .eventType(CreditEvent.CANCELLED)
                                            .creditId(saved.getId())
                                            .customerId(saved.getCustomerId())
                                            .status(saved.getStatus())
                                            .balance(saved.getBalance().toPlainString())
                                            .currency(saved.getCurrency())
                                            .occurredAt(OffsetDateTime.now(ZoneOffset.UTC))
                                            .correlationId(correlationId)
                                            .build())
                                    .thenReturn(saved));
                })
                .then();
    }

    @Override
    public Mono<OverdueInfo> getOverdueInfo(UUID creditId) {
        return findById(creditId)
                .map(doc -> {
                    if (doc.getDueDate() == null || doc.getBalance().compareTo(BigDecimal.ZERO) <= 0) {
                        return new OverdueInfo(false, BigDecimal.ZERO, 0L);
                    }
                    long days = ChronoUnit.DAYS.between(doc.getDueDate(),
                            OffsetDateTime.now(ZoneOffset.UTC).toLocalDate());
                    if (days <= 0) {
                        return new OverdueInfo(false, BigDecimal.ZERO, 0L);
                    }
                    return new OverdueInfo(true, doc.getBalance(), days);
                });
    }

    private Mono<Void> validateType(String type) {
        if (!TYPE_PERSONAL.equals(type) && !TYPE_BUSINESS.equals(type)) {
            return Mono.error(new CreditConflictException("Invalid credit type: " + type));
        }
        return Mono.empty();
    }
}

package com.banking.credit.service.impl;

import com.banking.credit.document.CreditDocument;
import com.banking.credit.exception.CreditNotFoundException;
import com.banking.credit.exception.PaymentExceedsBalanceException;
import com.banking.credit.exception.TransactionServiceException;
import com.banking.credit.repository.CreditRepository;
import com.banking.credit.dto.OperationAcceptedResponse;
import com.banking.credit.service.PaymentService;
import java.math.BigDecimal;
import java.time.OffsetDateTime;
import java.time.ZoneOffset;
import java.util.HashMap;
import java.util.Map;
import java.util.UUID;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.MediaType;
import org.springframework.stereotype.Service;
import org.springframework.web.reactive.function.client.WebClient;
import reactor.core.publisher.Mono;

/**
 * Fachada de pagos: delega la operacion en ms-transaction-service.
 * - Registra idempotencia por X-Transaction-ID (repeticion no genera doble efecto).
 * - Timeout de Resilience4j (2 s) aplicado en la llamada saliente.
 * - El saldo/estado del credito se actualiza por el flujo de eventos/transaction-service,
 *   sin escrituras cruzadas a otras bases.
 */
@Slf4j
@Service
@RequiredArgsConstructor
public class PaymentServiceImpl implements PaymentService {

    private final CreditRepository creditRepository;
    private final WebClient transactionWebClient;

    @Value("${transaction-service.base-url:http://ms-transaction-service}")
    private String baseUrl;

    /** Registro de idempotencia en memoria por transaccion (transactionId -> respuesta). */
    private final Map<UUID, OperationAcceptedResponse> idempotencyRegistry = new HashMap<>();

    @Override
    public Mono<OperationAcceptedResponse> pay(UUID creditId, UUID transactionId,
                                               BigDecimal amount, String currency,
                                               UUID sourceAccountId, UUID payerCustomerId,
                                               String correlationId) {
        // 1) Idempotencia: mismo X-Transaction-ID retorna la transaccion original.
        synchronized (idempotencyRegistry) {
            var previous = idempotencyRegistry.get(transactionId);
            if (previous != null) {
                log.info("Idempotent replay for transactionId={}", transactionId);
                return Mono.just(previous);
            }
        }

        // 2) Validar credito y deuda pendiente.
        return creditRepository.findById(creditId)
                .switchIfEmpty(Mono.error(new CreditNotFoundException(
                        "Credit not found: " + creditId)))
                .flatMap(credit -> validate(credit, amount, currency))
                .flatMap(credit -> callTransactionService(creditId, transactionId, amount,
                        currency, sourceAccountId, payerCustomerId, correlationId));
    }

    private Mono<CreditDocument> validate(CreditDocument credit, BigDecimal amount, String currency) {
        if (!currency.equals(credit.getCurrency())) {
            return Mono.error(new PaymentExceedsBalanceException(
                    "Payment currency " + currency + " does not match credit currency "
                            + credit.getCurrency()));
        }
        if (amount.compareTo(credit.getBalance()) > 0) {
            return Mono.error(new PaymentExceedsBalanceException(
                    "Payment " + amount + " exceeds outstanding balance "
                            + credit.getBalance()));
        }
        return Mono.just(credit);
    }

    private Mono<OperationAcceptedResponse> callTransactionService(
            UUID creditId, UUID transactionId, BigDecimal amount, String currency,
            UUID sourceAccountId, UUID payerCustomerId, String correlationId) {

        Map<String, Object> payload = new HashMap<>();
        payload.put("creditId", creditId);
        payload.put("transactionId", transactionId);
        payload.put("amount", amount.toPlainString());
        payload.put("currency", currency);
        payload.put("sourceAccountId", sourceAccountId);
        if (payerCustomerId != null) {
            payload.put("payerCustomerId", payerCustomerId);
        }

        return transactionWebClient.post()
                .uri("/api/v1/transactions/credits/payments")
                .header("X-Correlation-ID", correlationId)
                .header("X-Transaction-ID", transactionId.toString())
                .contentType(MediaType.APPLICATION_JSON)
                .bodyValue(payload)
                .retrieve()
                .onStatus(status -> status.value() >= 400, response ->
                        response.bodyToMono(String.class)
                                .defaultIfEmpty("")
                                .flatMap(body -> Mono.error(new TransactionServiceException(
                                        "Transaction service rejected payment: "
                                                + response.statusCode() + " " + body))))
                .bodyToMono(OperationAcceptedResponse.class)
                .timeout(java.time.Duration.ofSeconds(2))
                .doOnNext(resp -> {
                    synchronized (idempotencyRegistry) {
                        idempotencyRegistry.put(transactionId, resp);
                    }
                    log.info("Payment delegated for credit {} [transactionId={}, status={}]",
                            creditId, transactionId, resp.getStatus());
                })
                .onErrorMap(err -> err instanceof TransactionServiceException ? err
                        : new TransactionServiceException(
                        "Failed to reach transaction service: " + err.getMessage()));
    }
}
