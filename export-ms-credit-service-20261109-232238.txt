package com.banking.credit.config;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.config.annotation.web.reactive.EnableWebFluxSecurity;
import org.springframework.security.config.web.server.ServerHttpSecurity;
import org.springframework.security.web.server.SecurityWebFilterChain;

@Configuration
@EnableWebFluxSecurity
public class SecurityConfig {

    @Value("${spring.security.dev-mode:false}")
    private boolean devMode;

    @Bean
    public SecurityWebFilterChain securityWebFilterChain(ServerHttpSecurity http) {
        if (devMode) {
            return http
                    .csrf(ServerHttpSecurity.CsrfSpec::disable)
                    .authorizeExchange(exchanges -> exchanges.anyExchange().permitAll())
                    .build();
        }
        return http
                .csrf(ServerHttpSecurity.CsrfSpec::disable)
                .authorizeExchange(exchanges -> exchanges.anyExchange().authenticated())
                .oauth2ResourceServer(oauth2 -> oauth2.jwt(jwt -> { }))
                .build();
    }
}

package com.banking.credit.controller;

import com.banking.credit.api.CreditApi;
import com.banking.credit.dto.CreditCreateRequest;
import com.banking.credit.dto.CreditPage;
import com.banking.credit.dto.CreditResponse;
import com.banking.credit.dto.CreditUpdateRequest;
import com.banking.credit.mapper.CreditMapper;
import com.banking.credit.service.CreditService;
import java.math.BigDecimal;
import lombok.RequiredArgsConstructor;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.RestController;
import org.springframework.web.server.ServerWebExchange;
import reactor.core.publisher.Mono;

import java.util.UUID;

@RestController
@RequiredArgsConstructor
public class CreditApiImpl implements CreditApi {

    private final CreditService creditService;
    private final CreditMapper creditMapper;

    @Override
    public Mono<ResponseEntity<CreditPage>> findAllCredits(String xCorrelationID, String xChannelID,
                                                           Integer page, Integer size,
                                                           ServerWebExchange exchange) {
        return creditService.findAll(page, size)
                .map(creditMapper::toResponse)
                .collectList()
                .map(items -> {
                    var p = new CreditPage();
                    p.setContent(items);
                    p.setPage(page);
                    p.setSize(size);
                    p.setTotalElements(items.size());
                    return ResponseEntity.ok(p);
                });
    }

    @Override
    public Mono<ResponseEntity<CreditResponse>> createCredit(String xCorrelationID, String xChannelID,
                                                             UUID xTransactionID,
                                                             Mono<CreditCreateRequest> body,
                                                             ServerWebExchange exchange) {
        return body.flatMap(req -> creditService.create(req.getCustomerId(),
                        req.getType().getValue(), new BigDecimal(req.getAmount()),
                        req.getCurrency(), req.getTermMonths(), xCorrelationID))
                .map(resp -> ResponseEntity.status(HttpStatus.CREATED).body(resp));
    }

    @Override
    public Mono<ResponseEntity<CreditResponse>> findCreditById(UUID creditId, String xCorrelationID,
                                                               String xChannelID,
                                                               ServerWebExchange exchange) {
        return creditService.findById(creditId)
                .map(creditMapper::toResponse)
                .map(ResponseEntity::ok);
    }

    @Override
    public Mono<ResponseEntity<CreditResponse>> updateCredit(UUID creditId, String xCorrelationID,
                                                             String xChannelID, UUID xTransactionID,
                                                             Mono<CreditUpdateRequest> body,
                                                             ServerWebExchange exchange) {
        return body.flatMap(req -> creditService.update(creditId, req, xCorrelationID))
                .map(creditMapper::toResponse)
                .map(ResponseEntity::ok);
    }
}

package com.banking.credit.event;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.kafka.core.reactive.ReactiveKafkaProducerTemplate;
import org.springframework.stereotype.Component;
import reactor.core.publisher.Mono;

@Slf4j
@Component
@RequiredArgsConstructor
public class CreditEventPublisher {

    public static final String TOPIC = "banking.credit.events.v1";

    private final ReactiveKafkaProducerTemplate<String, CreditEvent> kafkaTemplate;

    public Mono<Void> publish(CreditEvent event) {
        String creditId = event.getPayload().getCreditId().toString();
        return kafkaTemplate.send(TOPIC, creditId, event)
                .doOnSuccess(result -> log.info(
                        "Event {} published for credit {} [correlationId={}]",
                        event.getEventType(), creditId, event.getCorrelationId()))
                .doOnError(err -> log.error(
                        "Failed to publish event {} for credit {}: {}",
                        event.getEventType(), creditId, err.getMessage()))
                .onErrorResume(err -> Mono.empty())
                .then();
    }
}

@Configuration
public class KafkaConfig {

    @Bean

        props.put(ProducerConfig.ENABLE_IDEMPOTENCE_CONFIG, true);

        return new ReactiveKafkaProducerTemplate<>(SenderOptions.create(props));
    }

package com.banking.credit.event.factory.impl;

import com.banking.credit.document.CreditDocument;
import com.banking.credit.event.CreditEvent;
import com.banking.credit.event.factory.CreditEventFactory;
import java.time.Instant;
import java.time.ZoneOffset;
import java.util.UUID;
import org.springframework.stereotype.Component;

@Component
public class CreditEventFactoryImpl implements CreditEventFactory {

    @Override
    public CreditEvent insert(CreditDocument credit, String correlationId) {
        return build(CreditEvent.INSERT, credit, correlationId);
    }

    @Override
    public CreditEvent update(CreditDocument credit, String correlationId) {
        return build(CreditEvent.UPDATE, credit, correlationId);
    }

    @Override
    public CreditEvent delete(CreditDocument credit, String correlationId) {
        return build(CreditEvent.DELETE, credit, correlationId);
    }

    private CreditEvent build(String eventType, CreditDocument credit, String correlationId) {
        return CreditEvent.builder()
                .eventId(UUID.randomUUID())
                .eventType(eventType)
                .occurredAt(Instant.now())
                .correlationId(correlationId)
                .payload(CreditEvent.CreditPayload.builder()
                        .creditId(credit.getId())
                        .customerId(credit.getCustomerId())
                        .type(credit.getType())
                        .status(credit.getStatus())
                        .amount(credit.getAmount().toPlainString())
                        .balance(credit.getBalance().toPlainString())
                        .currency(credit.getCurrency())
                        .termMonths(credit.getTermMonths())
                        .dueDate(credit.getDueDate() == null ? null : credit.getDueDate().toString())
                        .createdAt(credit.getCreatedAt() == null ? null
                                : credit.getCreatedAt().toInstant(ZoneOffset.UTC).toString())
                        .updatedAt(credit.getUpdatedAt() == null ? null
                                : credit.getUpdatedAt().toInstant(ZoneOffset.UTC).toString())
                        .build())
                .build();
    }
}

package com.banking.credit.exception;

import com.banking.credit.dto.ErrorResponse;
import java.time.OffsetDateTime;
import java.util.List;
import lombok.extern.slf4j.Slf4j;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.bind.support.WebExchangeBindException;
import org.springframework.web.server.ServerWebExchange;
import reactor.core.publisher.Mono;

@Slf4j
@RestControllerAdvice
public class GlobalExceptionHandler {

    private static ErrorResponse build(String code, String message, String correlationId,
                                       List<String> details) {
        var e = new ErrorResponse();
        e.setCode(code);
        e.setMessage(message);
        e.setTimestamp(OffsetDateTime.now());
        e.setCorrelationId(correlationId);
        e.setDetails(details);
        return e;
    }

    private static String correlation(ServerWebExchange exchange) {
        var v = exchange.getRequest().getHeaders().getFirst("X-Correlation-ID");
        return v == null ? "" : v;
    }

    @ExceptionHandler(CreditNotFoundException.class)
    public Mono<ResponseEntity<ErrorResponse>> notFound(CreditNotFoundException ex,
                                                        ServerWebExchange exchange) {
        return Mono.just(ResponseEntity.status(HttpStatus.NOT_FOUND)
                .body(build(ex.getCode(), ex.getMessage(), correlation(exchange), null)));
    }

    @ExceptionHandler(CreditConflictException.class)
    public Mono<ResponseEntity<ErrorResponse>> conflict(CreditConflictException ex,
                                                        ServerWebExchange exchange) {
        return Mono.just(ResponseEntity.status(HttpStatus.CONFLICT)
                .body(build(ex.getCode(), ex.getMessage(), correlation(exchange), null)));
    }

    @ExceptionHandler(WebExchangeBindException.class)
    public Mono<ResponseEntity<ErrorResponse>> badRequest(WebExchangeBindException ex,
                                                          ServerWebExchange exchange) {
        var details = ex.getBindingResult().getFieldErrors().stream()
                .map(fe -> fe.getField() + ": " + fe.getDefaultMessage())
                .toList();
        return Mono.just(ResponseEntity.badRequest()
                .body(build("BAD_REQUEST", "Invalid request", correlation(exchange), details)));
    }

    @ExceptionHandler(Exception.class)
    public Mono<ResponseEntity<ErrorResponse>> generic(Exception ex, ServerWebExchange exchange) {
        log.error("Unhandled error", ex);
        return Mono.just(ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR)
                .body(build("INTERNAL_ERROR", "Internal server error", correlation(exchange), null)));
    }
}

package com.banking.credit.mapper;

import com.banking.credit.document.CreditDocument;
import com.banking.credit.dto.CreditResponse;
import java.math.BigDecimal;
import java.time.OffsetDateTime;
import java.time.ZoneOffset;
import java.util.UUID;
import org.springframework.stereotype.Component;

@Component
public class CreditMapper {

    public CreditDocument toNewDocument(UUID customerId, String type, BigDecimal amount,
                                        String currency, int termMonths) {
        var now = OffsetDateTime.now(ZoneOffset.UTC).toLocalDateTime();
        return CreditDocument.builder()
                .id(UUID.randomUUID())
                .customerId(customerId)
                .type(type)
                .status(CreditDocument.STATUS_ACTIVE)
                .amount(amount)
                .balance(amount)
                .currency(currency)
                .termMonths(termMonths)
                .dueDate(OffsetDateTime.now(ZoneOffset.UTC).toLocalDate().plusMonths(termMonths))
                .createdAt(now)
                .updatedAt(now)
                .build();
    }

    public CreditResponse toResponse(CreditDocument doc) {
        var r = new CreditResponse();
        r.setId(doc.getId());
        r.setCustomerId(doc.getCustomerId());
        r.setType(com.banking.credit.dto.CreditType.fromValue(doc.getType()));
        r.setStatus(com.banking.credit.dto.CreditStatus.fromValue(doc.getStatus()));
        r.setAmount(doc.getAmount().toPlainString());
        r.setBalance(doc.getBalance().toPlainString());
        r.setCurrency(doc.getCurrency());
        r.setDueDate(doc.getDueDate());
        r.setCreatedAt(doc.getCreatedAt() == null ? null
                : doc.getCreatedAt().atOffset(ZoneOffset.UTC));
        return r;
    }
}

package com.banking.credit.service.impl;

import com.banking.credit.document.CreditDocument;
import com.banking.credit.dto.CreditResponse;
import com.banking.credit.dto.CreditUpdateRequest;
import com.banking.credit.event.CreditEventPublisher;
import com.banking.credit.event.factory.CreditEventFactory;
import com.banking.credit.exception.CreditNotFoundException;
import com.banking.credit.mapper.CreditMapper;
import com.banking.credit.repository.CreditRepository;
import com.banking.credit.service.CreditService;
import com.banking.credit.service.validator.CreditCreationValidator;
import com.banking.credit.service.validator.CreditUpdateValidator;
import java.math.BigDecimal;
import java.time.OffsetDateTime;
import java.time.ZoneOffset;
import java.util.UUID;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import reactor.core.publisher.Flux;
import reactor.core.publisher.Mono;

@Slf4j
@Service
@RequiredArgsConstructor
public class CreditServiceImpl implements CreditService {

    public static final String TYPE_PERSONAL = "PERSONAL";

    private final CreditRepository creditRepository;
    private final CreditMapper creditMapper;
    private final CreditEventPublisher eventPublisher;
    private final CreditEventFactory eventFactory;
    private final CreditCreationValidator creationValidator;
    private final CreditUpdateValidator updateValidator;

    @Override
    public Mono<CreditResponse> create(UUID customerId, String type, BigDecimal amount,
                                       String currency, Integer termMonths, String correlationId) {
        return creditRepository.existsByCustomerIdAndType(customerId, type)
                .flatMap(exists -> creationValidator.validate(type, Boolean.TRUE.equals(exists)))
                .then(Mono.defer(() -> creditRepository.save(
                        creditMapper.toNewDocument(customerId, type, amount, currency, termMonths))))
                .flatMap(saved -> eventPublisher.publish(eventFactory.insert(saved, correlationId))
                        .thenReturn(saved))
                .map(creditMapper::toResponse);
    }

    @Override
    public Flux<CreditDocument> findAll(int page, int size) {
        return creditRepository.findAll()
                .skip((long) page * size)
                .take(size);
    }

    @Override
    public Mono<CreditDocument> findById(UUID creditId) {
        return creditRepository.findById(creditId)
                .switchIfEmpty(Mono.error(new CreditNotFoundException(
                        "Credit not found: " + creditId)));
    }

    @Override
    public Mono<CreditDocument> update(UUID creditId, CreditUpdateRequest request, String correlationId) {
        return findById(creditId)
                .flatMap(doc -> updateValidator.validate(doc, request).thenReturn(doc))
                .flatMap(doc -> {
                    if (request == null || request.getStatus() == null) {
                        return Mono.just(doc);
                    }
                    doc.setStatus(request.getStatus().getValue());
                    doc.setUpdatedAt(OffsetDateTime.now(ZoneOffset.UTC).toLocalDateTime());
                    return creditRepository.save(doc)
                            .flatMap(saved -> eventPublisher.publish(eventFactory.update(saved, correlationId))
                                    .thenReturn(saved));
                });
    }
}

import com.banking.credit.exception.CreditConflictException;
import com.banking.credit.service.validator.CreditCreationValidator;
import org.springframework.stereotype.Component;
import reactor.core.publisher.Mono;

@Component
public class CreditCreationValidatorImpl implements CreditCreationValidator {

    private static final String TYPE_PERSONAL = "PERSONAL";
    private static final String TYPE_BUSINESS = "BUSINESS";

    @Override
    public Mono<Void> validate(String type, boolean existsPersonal) {
        if (!TYPE_PERSONAL.equals(type) && !TYPE_BUSINESS.equals(type)) {
            return Mono.error(new CreditConflictException("Invalid credit type: " + type));
        }
        if (TYPE_PERSONAL.equals(type) && existsPersonal) {
            return Mono.error(new CreditConflictException(
                    "Customer already has a PERSONAL credit"));
        }
        return Mono.empty();
    }
}

package com.banking.credit.service.validator.impl;

import com.banking.credit.document.CreditDocument;
import com.banking.credit.dto.CreditUpdateRequest;
import com.banking.credit.exception.CreditConflictException;
import com.banking.credit.service.validator.CreditUpdateValidator;
import java.math.BigDecimal;
import org.springframework.stereotype.Component;
import reactor.core.publisher.Mono;

@Component
public class CreditUpdateValidatorImpl implements CreditUpdateValidator {

    @Override
    public Mono<Void> validate(CreditDocument current, CreditUpdateRequest request) {
        if (request == null || request.getStatus() == null) {
            return Mono.empty();
        }
        String newStatus = request.getStatus().getValue();
        if (CreditDocument.STATUS_PAID.equals(newStatus)
                && current.getBalance().compareTo(BigDecimal.ZERO) > 0) {
            return Mono.error(new CreditConflictException(
                    "Cannot set PAID while balance is greater than zero"));
        }
        return Mono.empty();
    }
}

import org.junit.jupiter.api.Test;
import reactor.test.StepVerifier;

class CreditCreationValidatorImplTest {

    private final CreditCreationValidatorImpl validator = new CreditCreationValidatorImpl();

    @Test
    void personalFirstCreditIsAllowed() {
        StepVerifier.create(validator.validate("PERSONAL", false))
                .verifyComplete();
    }

    @Test
    void personalSecondCreditIsRejected() {
        StepVerifier.create(validator.validate("PERSONAL", true))
                .expectErrorMatches(e -> e instanceof CreditConflictException
                        && e.getMessage().contains("PERSONAL"))
                .verify();
    }

    @Test
    void businessIsAlwaysAllowed() {
        StepVerifier.create(validator.validate("BUSINESS", true))
                .verifyComplete();
        StepVerifier.create(validator.validate("BUSINESS", false))
                .verifyComplete();
    }

    @Test
    void invalidTypeIsRejected() {
        StepVerifier.create(validator.validate("MORTGAGE", false))
                .expectErrorMatches(e -> e instanceof CreditConflictException
                        && e.getMessage().contains("Invalid credit type"))
                .verify();
    }

    @Test
    void nullTypeIsRejected() {
        StepVerifier.create(validator.validate(null, false))
                .expectError(CreditConflictException.class)
                .verify();
    }
}

package com.banking.credit.service.validator.impl;

import com.banking.credit.document.CreditDocument;
import com.banking.credit.dto.CreditStatus;
import com.banking.credit.dto.CreditUpdateRequest;
import com.banking.credit.exception.CreditConflictException;
import java.math.BigDecimal;
import org.junit.jupiter.api.Test;
import reactor.test.StepVerifier;

class CreditUpdateValidatorImplTest {

    private final CreditUpdateValidatorImpl validator = new CreditUpdateValidatorImpl();

    private CreditDocument doc(BigDecimal balance) {
        return CreditDocument.builder()
                .id(java.util.UUID.randomUUID())
                .balance(balance)
                .status("ACTIVE")
                .build();
    }

    private CreditUpdateRequest req(CreditStatus status) {
        var r = new CreditUpdateRequest();
        r.setStatus(status);
        return r;
    }

    @Test
    void nullRequestIsAllowed() {
        StepVerifier.create(validator.validate(doc(BigDecimal.TEN), null))
                .verifyComplete();
    }

    @Test
    void requestWithoutStatusIsAllowed() {
        StepVerifier.create(validator.validate(doc(BigDecimal.TEN), new CreditUpdateRequest()))
                .verifyComplete();
    }

    @Test
    void paidWithPositiveBalanceIsRejected() {
        StepVerifier.create(validator.validate(doc(new BigDecimal("0.01")), req(CreditStatus.PAID)))
                .expectError(CreditConflictException.class)
                .verify();
    }

    @Test
    void paidWithZeroBalanceIsAllowed() {
        StepVerifier.create(validator.validate(doc(BigDecimal.ZERO), req(CreditStatus.PAID)))
                .verifyComplete();
    }

    @Test
    void cancelledWithPositiveBalanceIsAllowed() {
        StepVerifier.create(validator.validate(doc(new BigDecimal("500.00")),
                        req(CreditStatus.CANCELLED)))
                .verifyComplete();
    }

    @Test
    void overdueWithPositiveBalanceIsAllowed() {
        StepVerifier.create(validator.validate(doc(new BigDecimal("500.00")),
                        req(CreditStatus.OVERDUE)))
                .verifyComplete();
    }

    @Test
    void activeStatusIsAlwaysAllowed() {
        StepVerifier.create(validator.validate(doc(new BigDecimal("500.00")),
                        req(CreditStatus.ACTIVE)))
                .verifyComplete();
    }
}
